\section{Weighted score inequalities and finite-copy scaling}
\label{supp:score}

Let $P=\supp\rhozero$ and let $Z=ZP$ be right-supported on the baseline support. For a POVM effect $M_y\succeq0$, define $p_y=\Tr(\rhozero M_y)$.

\begin{lemma}[Weighted Hilbert--Schmidt score bound]
\label{lem:weighted-score}
For every outcome with $p_y>0$,
\begin{equation}
\frac{|\Tr(ZM_y)|^2}{p_y}
\le
\Tr(M_y Z\rhozero^+ Z^\dagger).
\label{eq:weighted-score}
\end{equation}
Consequently,
\begin{equation}
\sum_y\frac{|\Tr(ZM_y)|^2}{p_y}
\le
\Tr(Z\rhozero^+Z^\dagger).
\end{equation}
\end{lemma}

\begin{proof}
Set $X=M_y^{1/2}\rhozero^{1/2}$ and $Y=M_y^{1/2}Z\rhozero^{+1/2}$. Since $Z=ZP$,
\begin{equation}
\Tr(X^\dagger Y)=\Tr(M_yZ).
\end{equation}
Hilbert--Schmidt Cauchy--Schwarz gives
\begin{equation}
|\Tr(M_yZ)|^2
\le
\Tr(M_y\rhozero)\,
\Tr(M_yZ\rhozero^+Z^\dagger).
\end{equation}
Summing uses $\sum_yM_y=\id$.
\end{proof}

\begin{lemma}[Independent-copy weighted norm]
\label{lem:copy-scaling}
Let $Z$ be traceless and right-supported on $P$. Define
\begin{equation}
Z_N=\sum_{j=1}^N
\rhozero^{\otimes(j-1)}\otimes Z\otimes
\rhozero^{\otimes(N-j)}.
\end{equation}
Then
\begin{equation}
\Tr(Z_N\rhozero_N^+Z_N^\dagger)
=N\Tr(Z\rhozero^+Z^\dagger).
\label{eq:copy-scaling}
\end{equation}
\end{lemma}

\begin{proof}
The diagonal copy terms give the one-copy quadratic form. For $j\neq l$, the product factorizes through $\Tr Z\,\Tr Z^\dagger=0$. This is a property of the encoded state tangent and does not assume a product measurement.
\end{proof}

\section{Robust affine radius and spectral survival}
\label{supp:robust}

For a positive-radius tangent define on $P$
\begin{equation}
B=\rhozero^{-1/2}A\rhozero^{-1/2}.
\end{equation}

\begin{lemma}[Numerical-radius characterization]
\label{lem:numerical-radius}
If $\Rlin>0$, then
\begin{equation}
\Rlin=\frac1{w(B)},
\qquad
\|B\|\le\frac2{\Rlin}.
\end{equation}
\end{lemma}

\begin{proof}
Writing $\beta=x-iy$, the affine disk is congruent to
\begin{equation}
\id+\frac12(\beta B+\beta^*B^\dagger).
\end{equation}
For a unit vector $|\psi\rangle$, minimizing over the phase of $\beta$ gives $1-|\beta||\langle\psi|B|\psi\rangle|$. Positivity over the entire radius-$R$ disk is therefore equivalent to $Rw(B)\le1$. The standard inequality $\|B\|\le2w(B)$ then gives the norm bound \cite{BhatiaJain2023}.
\end{proof}

\begin{proposition}[Arbitrary-baseline upper-tail bound]
\label{prop:nonstationary-tail}
Suppose $\Rlin>0$ and $P_UA=A$. Then for every finite $N$ and arbitrary collective POVM,
\begin{equation}
\frac{\Rlin^2}{4}\frac{\Tr F_N}{N}
\le\Tr(P_U\rhozero).
\label{eq:nonstationary-tail}
\end{equation}
No commutation between $\rhozero$ and the Hamiltonian is required.
\end{proposition}

\begin{proof}
Positive radius implies $A=PAP$. Lemma~\ref{lem:weighted-score} bounds the Fisher trace by $\Tr(A\rhozero^+A^\dagger)$. By Lemma~\ref{lem:numerical-radius},
\begin{equation}
A\rhozero^+A^\dagger
\preceq\frac4{\Rlin^2}\rhozero.
\end{equation}
The left side is supported inside $P_U$. Compress by $P_U$, take the trace, and use Lemma~\ref{lem:copy-scaling} for finite $N$.
\end{proof}

For an exact positive Bohr gap of a semibounded Hamiltonian, $P_U$ may be chosen inside the upper energy tail. Applying the same result to $A^\dagger$ gives the lower/domain endpoint form. Applying these two views to the signal and clock Hamiltonians gives the main-text autonomous dual-survival law.

\section{Second-order positive-semidefinite-cone curvature}
\label{supp:psd}

Consider a two-sided $C^2$ curve
\begin{equation}
\rho(\theta)=\rhozero+\theta D+\frac{\theta^2}{2}C+o(\theta^2)
\end{equation}
that is physical for sufficiently small positive and negative $\theta$. Write $P=\supp\rhozero$, $Q=\id-P$, $R=P\rhozero P$, and $K=QDP$.

\begin{lemma}[Second-order support-creation condition]
\label{lem:psd-curvature}
Two-sided positivity implies
\begin{equation}
QDQ=0,
\qquad
QCQ\succeq2KR^{-1}K^\dagger.
\label{eq:psd-curvature}
\end{equation}
\end{lemma}

\begin{proof}
In the decomposition $P\oplus Q$, the support block is $R+O(\theta)$ and the off-diagonal block is $\theta K+O(\theta^2)$. Positivity of the Schur complement gives
\begin{equation}
\frac{\theta^2}{2}QCQ-\theta^2KR^{-1}K^\dagger+o(\theta^2)\succeq0,
\end{equation}
which yields Eq.~\eqref{eq:psd-curvature}. This is established PSD-cone second-order geometry, not a novelty claim.
\end{proof}

Applying Lemma~\ref{lem:psd-curvature} to the cosine and sine coordinates separately gives the main-text one-sided bound $J(A|\rhozero)\le\Delta T_U(0)$.

\section{Bilateral boundary scores and shared curvature}
\label{supp:bilateral}

Assume $QAQ=0$ and define
\begin{equation}
X=AP,
\qquad
Y=QA^\dagger P,
\qquad
A=X+Y^\dagger.
\end{equation}
For a POVM define the score vectors
\begin{equation}
u_y=\frac{\Tr(XM_y)}{\sqrt{p_y}},\qquad
v_y=\frac{\Tr(YM_y)^*}{\sqrt{p_y}}.
\end{equation}
Lemma~\ref{lem:weighted-score} gives $\|u\|_2^2\le J_X$ and $\|v\|_2^2\le J_Y$. Therefore
\begin{equation}
\sqrt{\Tr F_1}=\|u+v\|_2\le\sqrt{J_X}+\sqrt{J_Y}.
\end{equation}
Lemma~\ref{lem:copy-scaling} gives the finite-copy version.

For a pure-boundary $C^2$ family define
\begin{equation}
Z_+=X\rhozero^+X^\dagger,
\qquad
Z_-=Y\rhozero^+Y^\dagger,
\end{equation}
and
\begin{equation}
C_\Delta=Q(\partial_x^2+\partial_y^2)\rho(0)Q.
\end{equation}
Summing the two quadrature versions of Lemma~\ref{lem:psd-curvature} cancels the $X/Y$ cross terms and gives
\begin{equation}
C_\Delta\succeq Z_++Z_-.
\label{eq:shared-curvature}
\end{equation}
This is the shared operator resource behind the boundary-action results.

\section{Shorted endpoint geometry for coherent support}
\label{supp:shorting}

For the support-preserving component $B=PAP$, define
\begin{equation}
Z_B^+=B\rhozero^+B^\dagger,
\quad
Z_B^-=B^\dagger\rhozero^+B,
\quad
R_B^\pm=\supp Z_B^\pm.
\end{equation}
Let $\Pi_{S,U},\Pi_{S,D},\Pi_{C,U},\Pi_{C,D}$ denote the minimal participating local endpoint projectors for the support-preserving exchange content. Define the support compressions
\begin{align}
S_{S,U}&=P\Pi_{S,U}P,&S_{S,D}&=P\Pi_{S,D}P,\\
S_{C,U}&=P\Pi_{C,U}P,&S_{C,D}&=P\Pi_{C,D}P.
\end{align}
The four shorting constants are
\begin{align}
\lambda_{S,U}&=\sup\{\lambda\ge0:S_{S,U}\succeq\lambda R_B^+\},\\
\lambda_{C,D}&=\sup\{\lambda\ge0:S_{C,D}\succeq\lambda R_B^+\},\\
\lambda_{S,D}&=\sup\{\lambda\ge0:S_{S,D}\succeq\lambda R_B^-\},\\
\lambda_{C,U}&=\sup\{\lambda\ge0:S_{C,U}\succeq\lambda R_B^-\}.
\end{align}
Let $R_B$ be the affine radius of the support-preserving sub-tangent and $T_{X,E}=\Tr(\Pi_{X,E}\rhozero)$. Then
\begin{align}
J_B^+&\le\frac{4T_{S,U}}{R_B^2\lambda_{S,U}},&
J_B^+&\le\frac{4T_{C,D}}{R_B^2\lambda_{C,D}},\\
J_B^-&\le\frac{4T_{S,D}}{R_B^2\lambda_{S,D}},&
J_B^-&\le\frac{4T_{C,U}}{R_B^2\lambda_{C,U}}.
\end{align}
Therefore the main-text pre-existing resource ceilings may be taken as
\begin{align}
a_+&=\min\!\left\{
\frac{4T_{S,U}}{R_B^2\lambda_{S,U}},
\frac{4T_{C,D}}{R_B^2\lambda_{C,D}}
\right\},\\
a_-&=\min\!\left\{
\frac{4T_{S,D}}{R_B^2\lambda_{S,D}},
\frac{4T_{C,U}}{R_B^2\lambda_{C,U}}
\right\}.
\label{eq:apm-short}
\end{align}
These constants are generalized principal-angle/shorting quantities. In the clean commuting-support limit they equal one and recover the ordinary endpoint-survival law.

The four-level counterexample in the research record has $J_B^+=1/2$ while omitting the geometric correction gives only $1/8$. The exact value $\lambda_{S,U}=1/4$ repairs the inequality, and a physical POVM violates the naive no-geometry resource ceiling. Thus the shorting geometry is operationally necessary.

\section{Exact shared-curvature allocation}
\label{supp:allocation}

Let $C\succeq0$ and let $R_+,R_-$ be synthesized output projectors. Define
\begin{equation}
\mathfrak A(C;R_+,R_-)=
\left\{(Z_+,Z_-):
Z_\pm\succeq0,
\ Z_\pm=R_\pm Z_\pm R_\pm,
\ Z_++Z_-\preceq C
\right\}.
\end{equation}
For $a\ge0$ define
\begin{equation}
\Phi_a(C;R_+,R_-)
=\sup_{(Z_+,Z_-)\in\mathfrak A}
\left[\sqrt{a+\Tr Z_+}+\sqrt{\Tr Z_-}\right]^2.
\end{equation}
Using
\begin{equation}
(\sqrt u+\sqrt v)^2
=\inf_{0<\eta<1}\left(\frac u\eta+\frac v{1-\eta}\right),
\end{equation}
and convex--concave minimax,
\begin{equation}
\Phi_a
=\inf_{0<\eta<1}
\left[
\frac a\eta+h_{1/\eta,1/(1-\eta)}(C;R_+,R_-)
\right],
\end{equation}
where
\begin{align}
h_{\alpha,\beta}
={}&\max_{(Z_+,Z_-)\in\mathfrak A}
\alpha\Tr Z_+ +\beta\Tr Z_-\\
={}&\min_{W\succeq0}\Tr(CW)
\end{align}
subject to
\begin{equation}
R_+WR_+\succeq\alpha R_+,
\qquad
R_-WR_-\succeq\beta R_-.
\end{equation}
The second line is the SDP dual. The optimization mathematics is prior infrastructure; its role here is to avoid double charging a shared physical curvature operator.

\section{Positive spectral action and the \texorpdfstring{$\Psi$}{Psi} envelope}
\label{supp:action}

Let $G\succeq0$ and suppose
\begin{equation}
g_+=\lambda_{\min}(R_+GR_+|_{R_+})>0,
\qquad
g_-=\lambda_{\min}(R_-GR_-|_{R_-})>0.
\end{equation}
From Eq.~\eqref{eq:shared-curvature},
\begin{equation}
g_+J_+ +g_-J_-
\le\Tr(GC_\Delta)=4\Aact_G^{(2)}.
\end{equation}
For $e=4\Aact_G^{(2)}$, $p=g_+$, $q=g_-$ and internal ceiling $a$, maximize
\begin{equation}
\left[\sqrt{a+j_+}+\sqrt{j_-}\right]^2
\end{equation}
subject to $pj_++qj_-\le e$. Direct one-variable optimization gives
\begin{equation}
\Psi_a(e;p,q)=
\begin{cases}
(\sqrt a+\sqrt{e/q})^2,&e\le ap^2/q,\\[1mm]
(e+pa)(p^{-1}+q^{-1}),&e\ge ap^2/q.
\end{cases}
\end{equation}
For $a=0$, the effective bilateral price is the harmonic combination $(p^{-1}+q^{-1})^{-1}$.

\section{Canonical autonomous endpoint-incidence operator}
\label{supp:canonical-action}

For one exact exchange tangent $A_\nu$, define
\begin{equation}
\Pi_{\rm out}=\supp(A_\nu A_\nu^\dagger),
\qquad
\Pi_{\rm in}=\supp(A_\nu^\dagger A_\nu).
\end{equation}
From $[H_X,A_\nu]=s_X\hbar\nu A_\nu$ it follows that
\begin{equation}
[H_X,A_\nu A_\nu^\dagger]=[H_X,A_\nu^\dagger A_\nu]=0,
\end{equation}
so the support projectors commute with both local Hamiltonians. The audited canonical clock-plus-signal endpoint-incidence operator is
\begin{equation}
G_{\rm ex}=2\hbar\nu\,Q(\Pi_{\rm out}+\Pi_{\rm in})Q.
\end{equation}
If a joint state is both a domain and range endpoint, it is counted twice because it carries two endpoint roles. The associated action is kinematic and kernel-resolved, not a total protocol-energy accounting.

\section{Rank-one curvature-metric angle}
\label{supp:rankone-angle}

For $R_+=|u\rangle\langle u|$ and $R_-=|v\rangle\langle v|$, short the available curvature to $S=\operatorname{span}\{u,v\}$ and write $C_S=\operatorname{Short}_S(C)$. Define
\begin{equation}
s_u=\frac1{\langle u|C_S^{-1}|u\rangle},
\quad
s_v=\frac1{\langle v|C_S^{-1}|v\rangle},
\end{equation}
\begin{equation}
c=\frac{|\langle u|C_S^{-1}|v\rangle|}
{\sqrt{\langle u|C_S^{-1}|u\rangle\langle v|C_S^{-1}|v\rangle}}.
\end{equation}
The exact allocation frontier is
\begin{equation}
y_{\max}(x)=s_v\frac{1-x/s_u}{1-(1-c^2)x/s_u}.
\end{equation}
This proves that ordinary Hilbert-space endpoint angles do not generally contain enough information to describe shared-curvature allocation.

\section{Multi-gap common Hessian and Fourier readout}
\label{supp:multigap}

For one common $2m$-parameter family, apply Lemma~\ref{lem:psd-curvature} to every cosine and sine coordinate. For mode $k$,
\begin{equation}
K_{k,c}=\frac{X_k+Y_k}{2},
\qquad
K_{k,s}=\frac{X_k-Y_k}{2i}.
\end{equation}
Adding the two PSD inequalities cancels the cross terms and gives
\begin{equation}
C_{\Delta,k}\succeq Z_{k,+}+Z_{k,-}.
\end{equation}
Summing over $k$ before applying any scalar cost gives
\begin{equation}
C_\Sigma\succeq\sum_k(Z_{k,+}+Z_{k,-}).
\end{equation}
This order of operations ensures that one positive operator $G$ charges the common Hessian only once.

In the clean fixed-total-energy star shell,
\begin{equation}
A_k=c_k(|k\rangle\!\langle0|+|0\rangle\!\langle-k|),
\end{equation}
and the exact common nonlinear family has
\begin{equation}
C_\Sigma=\sum_k c_k^2
(|-k\rangle\!\langle-k|+|k\rangle\!\langle k|).
\end{equation}
For $d=2m+1$, the DFT vectors
\begin{equation}
|v_j\rangle=\frac1{\sqrt d}\sum_{n=-m}^{m}e^{in2\pi j/d}|n\rangle
\end{equation}
give
\begin{align}
F_{x_kx_l}&=2c_k^2\delta_{kl},\\
F_{y_ky_l}&=2c_k^2\delta_{kl},\\
F_{x_ky_l}&=0.
\end{align}
Thus the same common record saturates every mode block and the complete clean frequency-weighted action sum.
